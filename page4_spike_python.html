python - buildhat_motors.py 
from buildhat import Motor, MotorPair
import time

# Port A: 좌측 모터, Port B: 우측 모터
motor_a = Motor('A')
motor_b = Motor('B')

# 1) 단일 모터 제어
motor_a.run_for_degrees(360, speed=50)
print("Motor A 위치:", motor_a.get_aposition())

# 2) MotorPair 동시 동기화 주행
pair = MotorPair('A', 'B')
pair.run_for_seconds(3, speedl=40, speedr=40) # 3초 전진
pair.run_for_degrees(180, speedl=40, speedr=-40) # 제자리 회전
pair.stop()
python - buildhat_distance.py 
from buildhat import DistanceSensor
import time

dist = DistanceSensor('C')
dist.on()

for _ in range(10):
    distance = dist.get_distance()
    print(f"측정 거리: {distance} cm")
    time.sleep(0.5)

dist.wait_for_distance_closer_than(10)
print("10cm 이내 접근 감지!")
dist.off()
python - buildhat_force.py 
from buildhat import ForceSensor

force_sensor = ForceSensor('D')

print("눌림 상태:", force_sensor.is_pressed)
print("가해진 힘(N):", force_sensor.get_force())

def handle_pressed(force):
    print(f"🔴 힘센서 클릭 감지! 힘: {force} N")

force_sensor.when_pressed = handle_pressed
force_sensor.wait_until_pressed()
python - buildhat_color.py 
from buildhat import ColorSensor
import time

color_sensor = ColorSensor('B')

for _ in range(10):
    color = color_sensor.get_color()
    reflected = color_sensor.get_reflected_light()
    print(f"색상: {color:8s} | 반사광: {reflected}%")
    time.sleep(0.5)
